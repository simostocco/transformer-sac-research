cmake_minimum_required(VERSION 3.20)
project(streaming_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(MKL_INCLUDE_DIR mkl_cblas.h HINTS ENV MKLROOT PATH_SUFFIXES include REQUIRED)
find_library(MKL_RT_LIBRARY NAMES mkl_rt HINTS ENV MKLROOT PATH_SUFFIXES lib lib/intel64 REQUIRED)
add_library(research_core STATIC native/src/numeric.cpp native/src/attestation.cpp)
target_include_directories(research_core PUBLIC native/include ${MKL_INCLUDE_DIR})
target_compile_options(research_core PRIVATE -ffp-contract=off -fno-fast-math)
target_link_libraries(research_core PUBLIC Threads::Threads OpenSSL::Crypto ${MKL_RT_LIBRARY} ${CMAKE_DL_LIBS})
enable_testing()
foreach(name core_test allocation_test queue_controls context_microbatch_test)
  add_executable(${name} native/tests/${name}.cpp)
  target_compile_options(${name} PRIVATE -ffp-contract=off -fno-fast-math)
  target_link_libraries(${name} PRIVATE research_core)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
add_executable(numeric_probe native/tests/numeric_probe.cpp)
target_compile_options(numeric_probe PRIVATE -ffp-contract=off -fno-fast-math)
target_link_libraries(numeric_probe PRIVATE research_core)
